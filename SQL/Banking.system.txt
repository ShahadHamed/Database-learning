create database BankSystem
Use BankSystem

--DDL

create table Branches (
    branch_id int primary key identity(1,1),
    b_address varchar(200),
    phone_number int
)

create table Employees (
    e_id int primary key,
    branch_id int,
    e_name varchar(50),
    position varchar(50),
    Foreign key (branch_id) references Branches(branch_id)
)

create table Customers (
    c_id int primary key,
    c_name varchar(50),
    c_address varchar(200),
    BOD date
)

create table Emp_Cust (
    e_id int,
    c_id int,
    processing_loans char(1), --Y or N
    opening_accounts char(1), --Y or N
    primary key (e_id, c_id),
    Foreign key (e_id) references Employees(e_id),
    Foreign key (c_id) references Customers(c_id)
)

create table Loans (
    l_id int primary key identity(100,5),
    e_id int,
    c_id int,
    l_type varchar(50),
    amount decimal(12,2),
    issue_date date,
    Foreign key (e_id) references Employees(e_id),
    Foreign key (c_id) references Customers(c_id)
)

create table Account (
    account_no int primary key,
    c_id int,
    balance decimal(12,2),
    DOC date,
    account_type varchar(20),  -- savings / checking
    Foreign key (c_id) references Customers(c_id)
)

create table Transactions (
    transaction_id int primary key,
    account_no int,
    amount decimal(12,2),
    date date,
    transaction_type varchar(20),  -- transfer / withdrawal / deposit
    foreign key (account_no) references Account(account_no)
)
