Create database  University
Use  University

--DDL
create table Student (
       S_id int primary key, 
	   F_name nvarchar(50) not null,
       L_name nvarchar(50) not null,
	   Phone_no int,
	   DOB date
)

create table Department (
       Department_id int primary key identity(1,1), 
	   D_name nvarchar(50) not null
)

create table Faculty (
       F_id int primary key, 
	   FName varchar(50) not null,
	   Mobile_no int,
	   Department nvarchar(50),
	   Salary int
)

create table Fac_student(
       S_id int,
	   F_id int,
	   foreign key (S_id) references Student(S_id),
	   foreign key (F_id) references Faculty(F_id),
	   primary key (S_id,F_id)
)

create table Couese (
       Course_id int primary key identity(10,1),
	   Course_name nvarchar(50),
	   Duration int
)

create table Dep_course (
       Department_id int,
	   Course_id int,
	   foreign key (Department_id) references Department(Department_id),
	   foreign key (Course_id) references Couese(Course_id),
	   primary key (Department_id,Course_id)
)
 
create table Stud_course (
       S_id int,
	   Course_id int,
	   foreign key (S_id) references Student(S_id),
	   foreign key (Course_id) references Couese(Course_id),
	   primary key (S_id,Course_id)
)

create table Hostel ( 
        Hostel_id int primary key identity(1,1),
		Hostel_name nvarchar(50),
		City nvarchar(50),
		States nvarchar(50),
		H_address nvarchar(100),
        Pin_code int,
		No_of_seat int,
)

create table Subjects (
       Subject_id int primary key identity(20,1),
	   Subject_name nvarchar(50),
	   F_id int,
	   foreign key (F_id) references Faculty(F_id),
)

create table Stud_subject (
       S_id int,
	   Subject_id int,
	   foreign key (S_id) references Student(S_id),
	   foreign key (Subject_id) references Subjects(Subject_id),
	   primary key (S_id,Subject_id)
)

create table Exams (
       Exam_code int primary key,
	   EDate date,
	   ETime time,
	   Room varchar(20),
	   Department_id int,
	   foreign key (Department_id) references Department(Department_id)
)

create table Stud_exams (
       S_id int,
	   Exam_code int,
	   foreign key (S_id) references Student(S_id),
	   foreign key (Exam_code) references Exams(Exam_code),
	   primary key (S_id,Exam_code)
)

alter table Student
	add Department_id int foreign key references Department(Department_id) --TO ADD extra FK

 alter table Student
	add  Hostel_id int foreign key references Hostel(Hostel_id)

alter table Student
    add GPA decimal(3,3)

alter table Student
      alter column GPA decimal (3,2)

alter table Student
    add Enrollment_date date

alter table Student
    add current_status varchar(20)

--7. Display students with GPA between 3.0 and 3.5

--1. Display the department ID, name, and the full name of the faculty managing it.
select Department_id, D_name, FName AS Faculty_Name
from Department Inner join Faculty
on Department = D_name

--2. Display each program's name and the name of the department offering it.
select c.Course_name, d.D_name AS Department_Name
from Couese c Inner join Dep_course dc
  on c.Course_id = dc.Course_id
Inner join Department d
  on dc.Department_id = d.Department_id

--3. Display the full student data and the full name of their faculty advisor.
select s.*, f.FName AS Advisor_Name
from Student s Inner join Fac_student fs
  on s.S_id = fs.S_id
Inner join Faculty f
  on fs.F_id = f.F_id

--4. Display class(Exam) IDs, course titles, and room locations for classes in buildings 'A' or 'L'.
select e.Exam_code, c.Course_name, e.Room
from Exams e Inner join Department d
  on e.Department_id = d.Department_id
 Inner join Dep_course dc
  on d.Department_id = dc.Department_id
 Inner join Couese c
  on dc.Course_id = c.Course_id
where e.Room LIKE 'A%' OR e.Room LIKE 'L%'

--5. Display full data about courses whose titles start with "I" (e.g., "Introduction to...").
select * from Couese
where Course_name LIKE 'D%'

--6. Display names of students in program(Department) ID 3 whose GPA is between 2.5 and 3.5.
select F_name+' '+ L_name AS Student_Name, GPA
from Student 
where Department_id = 3 AND GPA BETWEEN 2.5 AND 3.5

--7. Retrieve student names in the Engineering program who earned grades ≥ 90 in the "Database" course.
 
--8. Find names of students who are advised by "Dr. Ahmed Hassan".
select s.F_name + ' ' + s.L_name AS Student_Name
from Student s Inner join Fac_student fs
  on s.S_id = fs.S_id
 Inner join Faculty f
  on fs.F_id = f.F_id
where f.FName = 'Ahmed Ali'

--9. Retrieve each student's name and the titles of courses they are enrolled in, ordered by course title.
select s.F_name + ' ' + s.L_name AS Student_Name, C.Course_name
from Student S Inner join Stud_course SC
on S.S_id = SC.S_id
Inner join Couese C
on SC.Course_id = C.Course_id
Order by c.Course_name

--10. For each class in Building 'Main', retrieve class ID, course name, department name, and faculty name teaching the class.
select e.Exam_code, c.Course_name, d.D_name AS Department_Name, f.FName AS Faculty_Name
from Exams e Inner join Department d
  ON e.Department_id = d.Department_id
Inner join Dep_course dc
  ON d.Department_id = dc.Department_id
Inner join Couese c
  ON dc.Course_id = c.Course_id
Inner join Faculty f
  ON f.Department = d.D_name
where e.Room LIKE 'A%'

--11. Display all faculty members who manage any department.
select Faculty.* 
from Faculty Inner join Department
ON Department = D_name

--12. Display all students and their advisors' names, even if some students don’t have advisors yet.
select s.F_name + ' ' + s.L_name AS Student_Name, f.FName AS Advisor_Name
from Student s LEFT JOIN Fac_student fs
  ON s.S_id = fs.S_id
LEFT JOIN Faculty f
  ON fs.F_id = f.F_id
