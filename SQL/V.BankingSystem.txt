create database BankingSystem_v
use BankingSystem_v

CREATE TABLE Customer (
CustomerID INT PRIMARY KEY,
FullName NVARCHAR(100),
Email NVARCHAR(100),
Phone NVARCHAR(15),
SSN CHAR(9)
)

CREATE TABLE Account (
AccountID INT PRIMARY KEY,
CustomerID INT FOREIGN KEY REFERENCES Customer(CustomerID),
Balance DECIMAL(10, 2),
AccountType VARCHAR(50),
Status VARCHAR(20)
)

CREATE TABLE Transactions (
TransactionID INT PRIMARY KEY,
AccountID INT FOREIGN KEY REFERENCES Account(AccountID),
Amount DECIMAL(10, 2),
Type VARCHAR(10), -- Deposit, Withdraw
TransactionDate DATETIME
)

CREATE TABLE Loan (
LoanID INT PRIMARY KEY,
CustomerID INT FOREIGN KEY REFERENCES Customer(CustomerID),
LoanAmount DECIMAL(12, 2),
LoanType VARCHAR(50),
Status VARCHAR(20)
)

--View Creation Scenarios:

--=====================================================
-- 1. Customer Service View
-- Shows: Customer Name, Phone, Account Status
-- Hides: SSN, Balance
--=====================================================
create view VCustomerService
AS
select  c.FullName, c.Phone, a.Status AS AccountStatus
From Customer c Inner join Account a
on c.CustomerID=a.CustomerID

select * from VCustomerService


--=====================================================
-- 2. Finance Department View
-- Shows: Account ID, Balance, Account Type
--=====================================================
create view VFinanceDep
AS 
select AccountID, Balance, AccountType
from Account

select * from VFinanceDep


--=====================================================
-- 3. Loan Officer View
-- Shows: Loan details + CustomerID only
-- Hides: Customer name, phone, SSN
--=====================================================
create view VLoanOfficer
AS
select * from loan


select * from VLoanOfficer


-- =====================================
-- 4. Transaction Summary View
-- Shows: Last 30 days transactions only
-- Fields: AccountID, Amount, TransactionDate
-- =====================================
create view VTransaction
AS 
select AccountID, Amount, TransactionDate
from Transactions
where TransactionDate >= DATEADD(DAY, -30, GETDATE())


select * from VTransaction
