create database Airline
use Airline

--DDL
create table Airport (
       Airport_code varchar(10) primary key, 
	   A_name nvarchar(50) not null,
	   States nvarchar(50),
	   City nvarchar(50)
)

create table Flight(
       Flight_no int primary key identity(10,1),
	   Airline nvarchar(50) not null,
	   Weekdays nvarchar(20),
	   Restrictions nvarchar(100) 
)

create table Fare(
       Flight_no int,
	   code varchar (10),
	   Amount int,
	   foreign key (Flight_no) references Flight(Flight_no),
	   primary key (Flight_no)
)

create table Flight_leg(
       leg_no int primary key identity(1,1),
	   Schedulaed_dep_time time,
	   Schedulaed_arr_time time,
	   Airport_code varchar(10),
	   Flight_no int,
	   foreign key (Airport_code) references Airport(Airport_code),
	   foreign key (Flight_no) references Flight(Flight_no)
)

alter table Flight_leg                                 -- to change datatypes 
      alter column Schedulaed_dep_time DateTime

alter table Flight_leg                                 -- to change datatypes 
      alter column Schedulaed_arr_time DateTime

create table Airplane_type(
       Type_no int primary key, 
	   Airport_code varchar(10),
	   TypeName varchar(50),
	   company varchar(50),
	   Max_seats int,
	   foreign key (Airport_code) references Airport(Airport_code)
)

create table Airplane(
       Airplane_id int primary key identity(100,2),
	   Type_no int,
	   Total_no_seats int,
	   foreign key (Type_no) references Airplane_type(Type_no)
)

create table leg_instance(
       leg_id int primary key, 
	   leg_no int,
	   Airplane_id int,
	   Arrival_time time,
	   Departure_time time,
	   No_available_seats int,
	   foreign key (leg_no) references Flight_leg(leg_no),
	   foreign key (Airplane_id) references Airplane(Airplane_id)
)

alter table leg_instance     --add column 
      add leg_Date Date

create table Reservation(
       Custumar_id int primary key, 
	   leg_id int,
	   Cus_Name nvarchar(50),
	   phone int,
	   foreign key (leg_id) references leg_instance( leg_id)
)

--1. Display each flight leg's ID, schedule, and the name of the airplane assigned to it. 
select LI.leg_id, FL.schedulaed_dep_time, FL.schedulaed_arr_time, AT.TypeName
from leg_instance LI Inner join Flight_leg FL
on LI.leg_no=FL.leg_no
Inner join Airplane A
on A.Airplane_id=LI.Airplane_id
Inner join Airplane_type AT
on AT.Type_no=A.Type_no

--2. Display all flight numbers and the names of the departure and arrival airports. 
select F.Flight_no, A.A_name
from Flight F Inner join Flight_leg FL
on F.Flight_no=FL.Flight_no
Inner join Airport A
on A.Airport_code= FL.Airport_code

--3. Display all reservation data with the name and phone of the customer who made each booking. 
select R.Custumar_id, R.Cus_Name, R.Phone, R.leg_id
from reservation R Inner join  leg_instance li 
on R.leg_id = LI.leg_id
 
--4. Display IDs and locations of flights departing from 'MCT'. 
select FL.Flight_no , A.States+' '+A.city AS locations
from Flight_leg FL Inner join Airport A
on FL.Airport_code=A.Airport_code
where A.Airport_code='MCT01'

--5. Display full data of flights whose names start with 'O'.

--6. List customers who have bookings with total payment between 150 and 500. 
select R.Cus_Name, F.Amount AS Total_Payment
from Reservation R Inner join leg_instance LI
on R.leg_id=LI.leg_id
Inner join Flight_leg FL
on FL.leg_no=LI.leg_no
Inner join Fare F
on F.Flight_no=FL.Flight_no
where F.Amount between 150 AND 500
