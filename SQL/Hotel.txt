Create database Hotel 
Use Hotel

--DLL
create table Branches ( 
      branch_id int primary key identity(1,1),
      branch_name VARCHAR(100) NOT NULL,
      locations VARCHAR(100) NOT NULL
)

create table Staff (
      staff_id int primary key,
      branch_id int NOT NULL,
      s_name varchar(20) NOT NULL,
      job_title varchar(50),
      salary decimal(10,2),
      foreign key (branch_id) references Branches(branch_id)
)

create table Rooms (
      room_no int primary key identity(100,1),
      branch_id int NOT NULL,
      room_type varchar(50),
      nightly_rate decimal(10,2),
      room_availability bit default 1,
      foreign key (branch_id) references Branches(branch_id)
)

create table Customer (
      customer_id int primary key,
	  c_name varchar(20) NOT NULL,
	  phone int,
	  email varchar(100)
)

create table Booking (
      b_id int primary key identity(220,1), 
	  customer_id int,
	  check_in DateTime NOT NULL,
      check_out DateTime NOT NULL,
      foreign key (customer_id) references Customer(customer_id)
)

create table Room_Book (
      room_no int,
      b_id int,
	  foreign key (room_no) references Rooms(room_no),
	  foreign key (b_id) references Booking(b_id),
	  primary key (room_no,b_id)
)

create table Staff_Book (
      staff_id int,
      b_id int,
	  foreign key (staff_id) references Staff (staff_id),
	  foreign key (b_id) references Booking(b_id),
	  primary key (staff_id,b_id)
)

--1. Display hotel ID, name, and the name of its manager. 
select B.branch_id, B.branch_name, S.s_name AS Employee_name 
from Branches B Inner join Staff S
on B.branch_id=S.branch_id
where S.job_title = 'manager'

--2. Display hotel names and the rooms available under them. 
select B.branch_name, R.room_no, R.room_type
from branches B Inner join rooms R 
on B.branch_id = R.branch_id
where R.room_availability = 1

--3. Display guest data along with the bookings they made. 
select C.* , B.b_id AS Booking_id ,B.check_in , B.check_out 
from Customer C Inner join Booking B
on C.customer_id=B.customer_id

--4. Display bookings for hotels in 'Muscat'. 
select b.b_id, br.branch_name, br.locations
from booking b inner join room_book rb 
on b.b_id = rb.b_id
inner join rooms r 
on rb.room_no = r.room_no
inner join branches br 
on r.branch_id = br.branch_id
where br.locations in ('Muscat')

--5. Display all room records where room type starts with "S" (e.g., "Suite", "Single"). 
select * from rooms
where room_type like 's%'

--6. List guests who booked rooms priced between 150 and 1000. 
select c.c_name AS Guest_Name, r.nightly_rate
from customer c inner join booking b 
on c.customer_id = b.customer_id
inner join room_book rb 
on b.b_id = rb.b_id
inner join rooms r 
on rb.room_no = r.room_no
where r.nightly_rate between 150 and 1000

--7. Retrieve guest names who have bookings marked as 'Confirmed' in hotel "Hilton Downtown". 

--8. Find guests whose bookings were handled by staff member "Mona Ali". 
select distinct c.c_name AS Guest_Name
from customer c inner join booking b 
on c.customer_id = b.customer_id
inner join staff_book sb 
on b.b_id = sb.b_id
inner join staff s 
on sb.staff_id = s.staff_id
where s.s_name = 'Fatima Ali'

--9. Display each guest’s name and the rooms they booked, ordered by room type. 
select c.c_name  AS Guest_Name, r.room_type
from customer c inner join booking b 
on c.customer_id = b.customer_id
inner join room_book rb 
on b.b_id = rb.b_id
inner join rooms r 
on rb.room_no = r.room_no
order by r.room_type

--10. For each hotel in 'Cairo', display hotel ID, name, manager name, and contact info. 
select b.branch_id, b.branch_name, s.s_name as manager_name
from branches b inner join staff s 
on b.branch_id = s.branch_id
where b.locations = 'Muscat'
  and s.job_title = 'manager'

--11. Display all staff members who hold 'Manager' positions. 
select staff_id, s_name, branch_id
from staff
where job_title = 'manager'

--12. Display all guests and their reviews, even if some guests haven't submitted any reviews.

from customer c left join booking b 
on c.customer_id = b.customer_id
